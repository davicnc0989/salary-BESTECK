function dayCalc(w,r){
  const a=toMin(r.in), b=toMin(r.out), grace=w.noGrace?0:5;
  let ci;
  if(a<=SHIFT_START+grace||r.waive) ci=SHIFT_START; else ci=Math.ceil(a/30)*30;
  let co;
  const f=Math.floor(b/30)*30;
  co=(f+30-b<=grace)?f+30:f;
  if(r.waive) co=Math.max(co,Math.min(Math.ceil(b/30)*30,SHIFT_END));
  if(co<ci) co=ci;
  const worked=Math.max(0,(co-ci-LUNCH)/60);
  return {a:a,b:b,ci:ci,co:co,worked:worked,frac:Math.min(worked,FULL)/FULL,ot:Math.max(0,worked-FULL),late:ci>SHIFT_START&&ci>a};
}
function dayStatus(w,ds){
  const dow=parse(ds).getDay();
  if(dow===0) return {type:'sun'};
  const r=S.records[w.id+'|'+ds];
  if(r&&r.leave) return {type:'A',kind:'Leave'};
  if(r&&r.absent) return {type:'A',kind:'Absent'};
  if(r&&r.in&&r.out) return Object.assign({type:'P'},dayCalc(w,r));
  if(ds>=TK) return {type:'pending'};
  if(r&&(r.in||r.out)) return {type:'miss'};
  return {type:'A',kind:'Absent'};
}
const isAbs=(w,dt)=>dayStatus(w,dKey(dt)).type==='A';
function sundayCut(w,ds){
  const dt=parse(ds); let n=0;
  for(let i=1;i<=6;i++) if(isAbs(w,addDays(dt,-i))) n++;
  return n>=2||(isAbs(w,addDays(dt,-1))&&isAbs(w,addDays(dt,1)));
}
function summary(w,y,m,upto){
  const dim=new Date(y,m+1,0).getDate(), last=upto||dim, rate=w.salary/dim;
  let P=0,A=0,LV=0,SP=0,SC=0,frac=0,ot=0,late=0,pend=0;
  for(let d=1;d<=last;d++){
    const ds=ymd(y,m,d), st=dayStatus(w,ds);
    if(st.type==='sun'){ if(ds<=TK){ if(sundayCut(w,ds)) SC++; else SP++; } }
    else if(st.type==='P'){P++;frac+=st.frac;ot+=st.ot;if(st.late)late++;}
    else if(st.type==='A'){A++;if(st.kind==='Leave')LV++;}
    else pend++;
  }
  const dayPay=frac*rate, sunPay=SP*rate, otPay=ot*(rate/FULL)*S.otMult, gross=dayPay+sunPay+otPay;
  const mk=y+'-'+pad(m+1);
  const adv=S.advances.filter(a=>a.workerId===w.id&&a.date.slice(0,7)===mk).reduce((s,a)=>s+a.amount,0);
  return {rate:rate,P:P,A:A,LV:LV,SP:SP,SC:SC,ot:ot,late:late,pend:pend,dayPay:dayPay,sunPay:sunPay,otPay:otPay,gross:gross,adv:adv,net:gross-adv};
}

  const given=S.advances.filter(a=>a.workerId===w.id&&a.date.slice(0,7)===mk&&a.date<=V.adv.date).reduce((t,a)=>t+a.amount,0);

function dayInfo(x,ds){
  const dt=parse(ds), dim=new Date(dt.getFullYear(),dt.getMonth()+1,0).getDate(), rate=x.salary/dim, s=dayStatus(x,ds);
  const o={name:x.name,status:'',inT:'—',outT:'—',worked:0,ot:0,pay:0,pres:false,abs:false};
  if(s.type==='sun'){
    if(ds>TK) o.status='Sunday'; else if(sundayCut(x,ds)) o.status='Sunday (cut)'; else {o.status='Sunday (paid)';o.pay=rate;}
  } else if(s.type==='P'){
    o.status='Present';o.pres=true;
    o.inT=t12(s.ci)+(s.ci!==s.a?' (actual '+t12(s.a)+')':'');
    o.outT=t12(s.co)+(s.co!==s.b?' (actual '+t12(s.b)+')':'');
    o.worked=s.worked;o.ot=s.ot;o.pay=s.frac*rate+s.ot*(rate/FULL)*S.otMult;
  } else if(s.type==='A'){o.status=s.kind;o.abs=true;}
  else if(s.type==='miss') o.status='In/Out incomplete';
  else o.status='Pending';
  return o;
}
